name: ML Pipeline — Build

on:
  workflow_run:
    workflows: ["ML Pipeline — Test"]
    types: [completed]
    branches: [main]

env:
  PYTHON_VERSION: '3.11'
  AZURE_SUBSCRIPTION_ID: ${{ secrets.AZURE_SUBSCRIPTION_ID }}
  AZURE_RESOURCE_GROUP: 'fraud-ml-rg'
  AZURE_WORKSPACE_NAME: 'fraud-ml-workspace'
  MLFLOW_TRACKING_URI: ${{ secrets.MLFLOW_TRACKING_URI }}

jobs:
  build-environment:
    name: Build Reproducible Environment
    runs-on: ubuntu-latest
    if: ${{ github.event.workflow_run.conclusion == 'success' }}
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: ${{ env.PYTHON_VERSION }}

      - name: Build conda environment from lock file
        run: |
          conda env create -f environment.yml -n fraudshield-build
          conda activate fraudshield-build
          python -c "import xgboost; print(xgboost.__version__)"
          python -c "import sklearn; print(sklearn.__version__)"

      - name: Export exact environment snapshot
        run: |
          conda env export --no-builds > environment-exact.yml
          pip freeze > requirements-exact.txt

      - name: Upload environment artifacts
        uses: actions/upload-artifact@v4
        with:
          name: environment-snapshot
          path: |
            environment-exact.yml
            requirements-exact.txt

  train-model:
    name: Train Model on Azure ML Compute
    runs-on: ubuntu-latest
    needs: build-environment
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Azure Login
        uses: azure/login@v2
        with:
          creds: ${{ secrets.AZURE_CREDENTIALS }}

      - name: Install Azure ML CLI v2
        run: az extension add -n ml -y

      - name: Set Azure defaults
        run: |
          az account set --subscription ${{ env.AZURE_SUBSCRIPTION_ID }}
          az configure --defaults workspace=${{ env.AZURE_WORKSPACE_NAME }} group=${{ env.AZURE_RESOURCE_GROUP }}

      - name: Submit training job to Azure ML
        run: |
          az ml job create             --file pipelines/train-fraud-model.yml             --set inputs.training_data.path=azureml:fraud-training-data:3             --set inputs.model_name=fraud-detection-xgboost             --set inputs.seed=42             --query name -o tsv > job_name.txt

      - name: Wait for training completion
        run: |
          JOB_NAME=$(cat job_name.txt)
          az ml job show --name $JOB_NAME --query status -o tsv
          # Poll until Completed or Failed (max 2 hours)
          timeout 7200 bash -c '
            while true; do
              STATUS=$(az ml job show --name '$JOB_NAME' --query status -o tsv)
              echo "Job status: $STATUS"
              if [ "$STATUS" == "Completed" ]; then exit 0; fi
              if [ "$STATUS" == "Failed" ] || [ "$STATUS" == "Canceled" ]; then exit 1; fi
              sleep 60
            done
          '

      - name: Capture MLflow run ID
        run: |
          JOB_NAME=$(cat job_name.txt)
          az ml job show --name $JOB_NAME --query properties.mlflow_run_id -o tsv > mlflow_run_id.txt

      - name: Upload MLflow run ID
        uses: actions/upload-artifact@v4
        with:
          name: mlflow-run-id
          path: mlflow_run_id.txt

name: ML Pipeline — Validate

on:
  workflow_run:
    workflows: ["ML Pipeline — Build"]
    types: [completed]
    branches: [main]

env:
  AZURE_SUBSCRIPTION_ID: ${{ secrets.AZURE_SUBSCRIPTION_ID }}
  AZURE_RESOURCE_GROUP: 'fraud-ml-rg'
  AZURE_WORKSPACE_NAME: 'fraud-ml-workspace'
  MLFLOW_TRACKING_URI: ${{ secrets.MLFLOW_TRACKING_URI }}

  # ── VALIDATION THRESHOLDS (from Scenario Business Cost Model) ──
  THRESHOLD_AUC_MIN: '0.940'
  THRESHOLD_F1_MIN: '0.860'
  THRESHOLD_LATENCY_P99_MAX_MS: '120'
  THRESHOLD_FPR_MAX: '0.020'
  THRESHOLD_FNR_MAX: '0.100'
  THRESHOLD_SHAP_STABILITY_MIN: '0.850'

jobs:
  model-validation:
    name: Model Performance & Fairness Validation
    runs-on: ubuntu-latest
    if: ${{ github.event.workflow_run.conclusion == 'success' }}
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Install dependencies
        run: pip install -r requirements-dev.txt

      - name: Download MLflow run ID
        uses: actions/download-artifact@v4
        with:
          name: mlflow-run-id

      - name: Azure Login
        uses: azure/login@v2
        with:
          creds: ${{ secrets.AZURE_CREDENTIALS }}

      - name: Fetch model metrics from MLflow
        run: |
          RUN_ID=$(cat mlflow_run_id.txt/mlflow_run_id.txt)
          python scripts/fetch_mlflow_metrics.py             --run-id $RUN_ID             --output validation_results.json

      - name: Validate performance gates
        run: |
          python scripts/validate_gates.py             --metrics validation_results.json             --gates '{"auc_roc": "${{ env.THRESHOLD_AUC_MIN }}", "f1_score": "${{ env.THRESHOLD_F1_MIN }}", "latency_p99": "${{ env.THRESHOLD_LATENCY_P99_MAX_MS }}", "fpr": "${{ env.THRESHOLD_FPR_MAX }}", "fnr": "${{ env.THRESHOLD_FNR_MAX }}", "shap_stability": "${{ env.THRESHOLD_SHAP_STABILITY_MIN }}"}'

      - name: Validate fairness metrics
        run: |
          python scripts/validate_fairness.py             --metrics validation_results.json             --threshold 0.05

      - name: Validate model explainability (SHAP)
        run: |
          python scripts/validate_shap.py             --run-id $(cat mlflow_run_id.txt/mlflow_run_id.txt)             --min-stability ${{ env.THRESHOLD_SHAP_STABILITY_MIN }}

      - name: Generate validation report
        run: |
          python scripts/generate_validation_report.py             --metrics validation_results.json             --output validation_report.md

      - name: Upload validation report
        uses: actions/upload-artifact@v4
        with:
          name: validation-report
          path: validation_report.md

      - name: Post validation summary to PR
        if: github.event_name == 'pull_request'
        uses: actions/github-script@v7
        with:
          script: |
            const fs = require('fs');
            const report = fs.readFileSync('validation_report.md', 'utf8');
            github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo: context.repo.repo,
              body: `## Model Validation Report

${report}`
            });

name: ML Pipeline — Promote

on:
  workflow_run:
    workflows: ["ML Pipeline — Validate"]
    types: [completed]
    branches: [main]

env:
  AZURE_SUBSCRIPTION_ID: ${{ secrets.AZURE_SUBSCRIPTION_ID }}
  AZURE_RESOURCE_GROUP: 'fraud-ml-rg'
  AZURE_WORKSPACE_NAME: 'fraud-ml-workspace'
  MODEL_NAME: 'fraud-detection-xgboost'

jobs:
  promote-to-staging:
    name: Promote Model to Staging
    runs-on: ubuntu-latest
    if: ${{ github.event.workflow_run.conclusion == 'success' }}
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Azure Login
        uses: azure/login@v2
        with:
          creds: ${{ secrets.AZURE_CREDENTIALS }}

      - name: Install Azure ML CLI v2
        run: az extension add -n ml -y

      - name: Set Azure defaults
        run: |
          az account set --subscription ${{ env.AZURE_SUBSCRIPTION_ID }}
          az configure --defaults workspace=${{ env.AZURE_WORKSPACE_NAME }} group=${{ env.AZURE_RESOURCE_GROUP }}

      - name: Download MLflow run ID
        uses: actions/download-artifact@v4
        with:
          name: mlflow-run-id

      - name: Register model from training run
        run: |
          RUN_ID=$(cat mlflow_run_id.txt/mlflow_run_id.txt)
          az ml model create             --name ${{ env.MODEL_NAME }}             --version $(date +%s)             --path azureml://jobs/$RUN_ID/outputs/model             --type mlflow_model             --tags "stage=staging" "eval_passed=true" "promoted_by=ci-cd" "promoted_at=$(date -u +%Y-%m-%dT%H:%M:%SZ)"

      - name: Verify staging registration
        run: |
          LATEST=$(az ml model list --name ${{ env.MODEL_NAME }} --query "[?tags.stage=='Staging'] | [0].version" -o tsv)
          echo "Staging model version: $LATEST"
          if [ -z "$LATEST" ]; then exit 1; fi

  promote-to-production:
    name: Promote to Production (Manual Approval Required)
    runs-on: ubuntu-latest
    needs: promote-to-staging
    environment:
      name: production-model-promotion
      url: https://ml.azure.com/model/${{ env.MODEL_NAME }}
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Azure Login
        uses: azure/login@v2
        with:
          creds: ${{ secrets.AZURE_CREDENTIALS }}

      - name: Install Azure ML CLI v2
        run: az extension add -n ml -y

      - name: Set Azure defaults
        run: |
          az account set --subscription ${{ env.AZURE_SUBSCRIPTION_ID }}
          az configure --defaults workspace=${{ env.AZURE_WORKSPACE_NAME }} group=${{ env.AZURE_RESOURCE_GROUP }}

      - name: Get latest staging version
        id: get-staging
        run: |
          STAGING_VERSION=$(az ml model list --name ${{ env.MODEL_NAME }} --query "[?tags.stage=='Staging'] | [0].version" -o tsv)
          echo "staging_version=$STAGING_VERSION" >> $GITHUB_OUTPUT

      - name: Transition to Production
        run: |
          az ml model update             --name ${{ env.MODEL_NAME }}             --version ${{ steps.get-staging.outputs.staging_version }}             --set tags.stage=Production             --set tags.promoted_to_prod_at=$(date -u +%Y-%m-%dT%H:%M:%SZ)             --set tags.promoted_by=${{ github.actor }}

      - name: Update champion alias
        run: |
          # Set champion alias to new production version
          az ml model update             --name ${{ env.MODEL_NAME }}             --version ${{ steps.get-staging.outputs.staging_version }}             --set tags.alias=champion

      - name: Notify team
        uses: slackapi/slack-github-action@v1
        with:
          payload: |
            {
              "text": "Model ${{ env.MODEL_NAME }} v${{ steps.get-staging.outputs.staging_version }} promoted to Production by ${{ github.actor }}"
            }
        env:
          SLACK_WEBHOOK_URL: ${{ secrets.SLACK_WEBHOOK_URL }}

name: ML Pipeline — Deploy (Canary)

on:
  workflow_run:
    workflows: ["ML Pipeline — Promote"]
    types: [completed]
    branches: [main]

  workflow_dispatch:
    inputs:
      canary_percentage:
        description: 'Initial canary traffic percentage'
        required: true
        default: '10'
        type: choice
        options: ['5', '10', '20']

env:
  AZURE_SUBSCRIPTION_ID: ${{ secrets.AZURE_SUBSCRIPTION_ID }}
  AZURE_RESOURCE_GROUP: 'fraud-ml-rg'
  AZURE_WORKSPACE_NAME: 'fraud-ml-workspace'
  ENDPOINT_NAME: 'fraud-scoring-prod'
  MODEL_NAME: 'fraud-detection-xgboost'
  CANARY_DURATION_MINUTES: '30'

jobs:
  deploy-canary:
    name: Deploy Canary (Green)
    runs-on: ubuntu-latest
    if: ${{ github.event.workflow_run.conclusion == 'success' }}
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Azure Login
        uses: azure/login@v2
        with:
          creds: ${{ secrets.AZURE_CREDENTIALS }}

      - name: Install Azure ML CLI v2
        run: az extension add -n ml -y

      - name: Set Azure defaults
        run: |
          az account set --subscription ${{ env.AZURE_SUBSCRIPTION_ID }}
          az configure --defaults workspace=${{ env.AZURE_WORKSPACE_NAME }} group=${{ env.AZURE_RESOURCE_GROUP }}

      - name: Get production model version
        id: get-model
        run: |
          PROD_VERSION=$(az ml model list --name ${{ env.MODEL_NAME }} --query "[?tags.stage=='Production'] | [0].version" -o tsv)
          echo "model_version=$PROD_VERSION" >> $GITHUB_OUTPUT

      - name: Create green deployment
        run: |
          cat > green-deployment.yml << 'EOF'
          $schema: https://azuremlschemas.azureedge.net/latest/managedOnlineDeployment.schema.json
          name: green
          endpoint_name: ${{ env.ENDPOINT_NAME }}
          model: azureml:${{ env.MODEL_NAME }}:${{ steps.get-model.outputs.model_version }}
          instance_type: Standard_DS3_v2
          instance_count: 2
          request_settings:
            request_timeout_ms: 5000
            max_concurrent_requests_per_instance: 10
          liveness_probe:
            initial_delay_seconds: 30
            period_seconds: 10
            failure_threshold: 3
          readiness_probe:
            initial_delay_seconds: 10
            period_seconds: 10
            success_threshold: 1
          EOF

          az ml online-deployment create             --file green-deployment.yml             --resource-group ${{ env.AZURE_RESOURCE_GROUP }}             --workspace-name ${{ env.AZURE_WORKSPACE_NAME }}

      - name: Route canary traffic
        run: |
          CANARY_PCT=${{ github.event.inputs.canary_percentage || '10' }}
          BLUE_PCT=$((100 - CANARY_PCT))

          az ml online-endpoint update             --name ${{ env.ENDPOINT_NAME }}             --traffic "blue=$BLUE_PCT green=$CANARY_PCT"             --resource-group ${{ env.AZURE_RESOURCE_GROUP }}             --workspace-name ${{ env.AZURE_WORKSPACE_NAME }}

      - name: Canary health check (5 min warm-up + 25 min observation)
        run: |
          echo "Waiting 5 minutes for warm-up..."
          sleep 300

          echo "Starting 25-minute canary observation..."
          for i in {1..5}; do
            echo "=== Check $i/5 ==="

            # Check endpoint health
            az ml online-endpoint show               --name ${{ env.ENDPOINT_NAME }}               --query "{state: provisioning_state, traffic: traffic}"               --resource-group ${{ env.AZURE_RESOURCE_GROUP }}               --workspace-name ${{ env.AZURE_WORKSPACE_NAME }}

            # Check deployment health
            az ml online-deployment show               --name green               --endpoint-name ${{ env.ENDPOINT_NAME }}               --query "{state: provisioning_state, requests: scoring_uri}"               --resource-group ${{ env.AZURE_RESOURCE_GROUP }}               --workspace-name ${{ env.AZURE_WORKSPACE_NAME }}

            # Query Azure Monitor for latency (last 5 min)
            echo "Querying Azure Monitor for p99 latency..."
            # (In production, use Azure Monitor REST API or Log Analytics query)

            sleep 300
          done

      - name: Auto-promote or auto-rollback decision
        id: canary-decision
        run: |
          # This would query Azure Monitor metrics in production
          # For template, we simulate the decision logic

          # Query: p99 latency < 120ms AND error rate < 0.1% AND throughput > 25 txns/sec
          LATENCY_P99=$(python scripts/query_monitor.py --metric latency_p99 --minutes 25)
          ERROR_RATE=$(python scripts/query_monitor.py --metric error_rate --minutes 25)

          echo "Canary p99 latency: ${LATENCY_P99}ms"
          echo "Canary error rate: ${ERROR_RATE}%"

          if (( $(echo "$LATENCY_P99 < 120" | bc -l) )) && (( $(echo "$ERROR_RATE < 0.1" | bc -l) )); then
            echo "decision=promote" >> $GITHUB_OUTPUT
            echo "Canary passed. Ready for full promotion."
          else
            echo "decision=rollback" >> $GITHUB_OUTPUT
            echo "Canary failed. Triggering rollback."
          fi

  promote-full-traffic:
    name: Promote to 100% Traffic
    runs-on: ubuntu-latest
    needs: deploy-canary
    if: ${{ needs.deploy-canary.outputs.decision == 'promote' }}
    steps:
      - name: Azure Login
        uses: azure/login@v2
        with:
          creds: ${{ secrets.AZURE_CREDENTIALS }}

      - name: Install Azure ML CLI v2
        run: az extension add -n ml -y

      - name: Route 100% to green
        run: |
          az ml online-endpoint update             --name ${{ env.ENDPOINT_NAME }}             --traffic "blue=0 green=100"             --resource-group ${{ env.AZURE_RESOURCE_GROUP }}             --workspace-name ${{ env.AZURE_WORKSPACE_NAME }}

      - name: Verify traffic shift
        run: |
          sleep 60
          az ml online-endpoint show             --name ${{ env.ENDPOINT_NAME }}             --query "traffic"             --resource-group ${{ env.AZURE_RESOURCE_GROUP }}             --workspace-name ${{ env.AZURE_WORKSPACE_NAME }}

      - name: Delete blue deployment (cleanup)
        run: |
          sleep 300  # Wait 5 min for connection drain
          az ml online-deployment delete             --name blue             --endpoint-name ${{ env.ENDPOINT_NAME }}             --yes             --resource-group ${{ env.AZURE_RESOURCE_GROUP }}             --workspace-name ${{ env.AZURE_WORKSPACE_NAME }}

      - name: Rename green to blue (for next deployment cycle)
        run: |
          # In Azure ML, we keep "green" as the new "blue" for next cycle
          # Or create a new blue deployment with the same config
          echo "Deployment cycle complete. Green is now production."

      - name: Notify success
        uses: slackapi/slack-github-action@v1
        with:
          payload: |
            {
              "text": "Canary deployment successful! 100% traffic now on green. Blue deployment cleaned up."
            }
        env:
          SLACK_WEBHOOK_URL: ${{ secrets.SLACK_WEBHOOK_URL }}

7. WORKFLOW 6: ROLLBACK (ml-pipeline-rollback.yml)
--------------------------------------------------------------------------------

name: ML Pipeline — Rollback

on:
  workflow_dispatch:
    inputs:
      reason:
        description: 'Rollback reason'
        required: true
        type: string
      target_version:
        description: 'Target model version to rollback to (optional, defaults to previous)'
        required: false
        type: string

  # Auto-triggered by drift detection or monitoring alerts
  repository_dispatch:
    types: [auto-rollback-triggered]

env:
  AZURE_SUBSCRIPTION_ID: ${{ secrets.AZURE_SUBSCRIPTION_ID }}
  AZURE_RESOURCE_GROUP: 'fraud-ml-rg'
  AZURE_WORKSPACE_NAME: 'fraud-ml-workspace'
  ENDPOINT_NAME: 'fraud-scoring-prod'
  MODEL_NAME: 'fraud-detection-xgboost'

jobs:
  rollback:
    name: Emergency Rollback
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Azure Login
        uses: azure/login@v2
        with:
          creds: ${{ secrets.AZURE_CREDENTIALS }}

      - name: Install Azure ML CLI v2
        run: az extension add -n ml -y

      - name: Set Azure defaults
        run: |
          az account set --subscription ${{ env.AZURE_SUBSCRIPTION_ID }}
          az configure --defaults workspace=${{ env.AZURE_WORKSPACE_NAME }} group=${{ env.AZURE_RESOURCE_GROUP }}

      - name: Determine rollback target
        id: target
        run: |
          if [ -n "${{ github.event.inputs.target_version }}" ]; then
            TARGET_VERSION=${{ github.event.inputs.target_version }}
          else
            # Get the previous production version (not current)
            TARGET_VERSION=$(az ml model list --name ${{ env.MODEL_NAME }} --query "[?tags.stage=='Production'] | sort_by(@, &version) | [-2].version" -o tsv)
          fi
          echo "target_version=$TARGET_VERSION" >> $GITHUB_OUTPUT
          echo "Rolling back to version: $TARGET_VERSION"

      - name: Create rollback deployment (emergency blue)
        run: |
          cat > rollback-deployment.yml << 'EOF'
          $schema: https://azuremlschemas.azureedge.net/latest/managedOnlineDeployment.schema.json
          name: blue-rollback
          endpoint_name: ${{ env.ENDPOINT_NAME }}
          model: azureml:${{ env.MODEL_NAME }}:${{ steps.target.outputs.target_version }}
          instance_type: Standard_DS3_v2
          instance_count: 3
          request_settings:
            request_timeout_ms: 5000
            max_concurrent_requests_per_instance: 10
          EOF

          az ml online-deployment create             --file rollback-deployment.yml             --resource-group ${{ env.AZURE_RESOURCE_GROUP }}             --workspace-name ${{ env.AZURE_WORKSPACE_NAME }}

      - name: Emergency traffic shift (0% to failed deployment)
        run: |
          az ml online-endpoint update             --name ${{ env.ENDPOINT_NAME }}             --traffic "blue-rollback=100 green=0"             --resource-group ${{ env.AZURE_RESOURCE_GROUP }}             --workspace-name ${{ env.AZURE_WORKSPACE_NAME }}

      - name: Verify rollback health
        run: |
          sleep 60
          az ml online-endpoint show             --name ${{ env.ENDPOINT_NAME }}             --query "{state: provisioning_state, traffic: traffic}"             --resource-group ${{ env.AZURE_RESOURCE_GROUP }}             --workspace-name ${{ env.AZURE_WORKSPACE_NAME }}

      - name: Delete failed deployment
        run: |
          az ml online-deployment delete             --name green             --endpoint-name ${{ env.ENDPOINT_NAME }}             --yes             --resource-group ${{ env.AZURE_RESOURCE_GROUP }}             --workspace-name ${{ env.AZURE_WORKSPACE_NAME }}
